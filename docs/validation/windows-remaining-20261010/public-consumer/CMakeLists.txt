cmake_minimum_required(VERSION 3.20)
project(coin_windows_fronts LANGUAGES C CXX)
find_package(Coin REQUIRED)
find_package(CoinRender REQUIRED)
add_executable(coin_win32_campaign main.cpp)
target_compile_features(coin_win32_campaign PRIVATE cxx_std_17)
target_compile_definitions(coin_win32_campaign PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(coin_win32_campaign PRIVATE CoinRender::CoinRender Coin::Coin user32 gdi32)
add_executable(d3d12_removal_probe removal.cpp)
target_link_libraries(d3d12_removal_probe PRIVATE d3d12 dxgi)

if(MSVC)
 target_compile_options(coin_win32_campaign PRIVATE /EHsc)
 target_compile_options(d3d12_removal_probe PRIVATE /EHsc)
endif()
add_executable(coin_native_removal removal_coin.cpp)
target_compile_features(coin_native_removal PRIVATE cxx_std_17)
target_compile_definitions(coin_native_removal PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(coin_native_removal PRIVATE CoinRender::CoinRender Coin::Coin user32)
if(MSVC)
 target_compile_options(coin_native_removal PRIVATE /EHsc)
endif()
