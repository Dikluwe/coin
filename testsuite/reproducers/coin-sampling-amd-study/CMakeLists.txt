cmake_minimum_required(VERSION 3.18)
project(CoinNativeSamplingProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
set(COIN_SOURCE_DIR "" CACHE PATH "Unmodified Coin checkout")
set(COIN_BINARY_DIR "" CACHE PATH "Build of that checkout")
find_library(COIN_LIBRARY NAMES Coin PATHS "${COIN_BINARY_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(coin-native-sampling native_sampling.cpp)
target_compile_definitions(coin-native-sampling PRIVATE GL_GLEXT_PROTOTYPES)
target_include_directories(coin-native-sampling PRIVATE "${COIN_SOURCE_DIR}/include" "${COIN_BINARY_DIR}/include")
target_link_libraries(coin-native-sampling PRIVATE "${COIN_LIBRARY}" OpenGL::GL OpenGL::EGL)
add_executable(pure-egl-sampling pure_egl.cpp)
target_compile_definitions(pure-egl-sampling PRIVATE GL_GLEXT_PROTOTYPES)
target_link_libraries(pure-egl-sampling PRIVATE OpenGL::GL OpenGL::EGL)
find_package(X11 REQUIRED)
find_package(OpenGL REQUIRED COMPONENTS OpenGL GLX)
add_executable(glx-visual-probe glx_visual.cpp)
target_link_libraries(glx-visual-probe PRIVATE OpenGL::GLX X11::X11)
