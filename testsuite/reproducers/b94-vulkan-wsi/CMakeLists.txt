cmake_minimum_required(VERSION 3.16)
project(B94VulkanWsiProbe LANGUAGES CXX)
find_package(Vulkan REQUIRED)
find_package(X11 REQUIRED)
add_executable(B94VulkanWsiProbe wsi_probe.cpp)
target_compile_features(B94VulkanWsiProbe PRIVATE cxx_std_17)
target_link_libraries(B94VulkanWsiProbe PRIVATE Vulkan::Vulkan ${X11_LIBRARIES})
target_include_directories(B94VulkanWsiProbe PRIVATE ${X11_INCLUDE_DIR})
option(B94_WSI_ENABLE_GLX "Build the optional independent GLX interleave control" OFF)
if(B94_WSI_ENABLE_GLX)
  find_package(OpenGL REQUIRED)
  target_compile_definitions(B94VulkanWsiProbe PRIVATE B94_WSI_HAVE_GLX=1)
  target_link_libraries(B94VulkanWsiProbe PRIVATE OpenGL::GL)
endif()
